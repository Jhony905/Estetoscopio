// ----------------- Config audio (tomado y respetado de tu script) -----------------
#define SAMPLE_RATE     16000                       // 16 kHz
#define I2S_READ_BYTES  2048                        // bytes por i2s_read() (múltiplo de 4)
#define RECORD_SECONDS  10
#define CHANNELS_OUT    1

#define I2S_PIN_BCK   15
#define I2S_PIN_WS     2
#define I2S_PIN_DIN   39
#define I2S_PIN_DOUT  -1

#define SD_PIN_CLK  14
#define SD_PIN_CMD  17
#define SD_PIN_D0   16

// WAV / gain
static float g_mic_gain = 3.0f;     // editable desde UI

const char *FILENAME = "/record_test.wav";

// ----------------- LVGL / Display -----------------
#define LVGL_HOR_RES EXAMPLE_LCD_WIDTH
#define LVGL_VER_RES EXAMPLE_LCD_HEIGHT

static lv_disp_t *disp;
static lv_indev_t *indev_touch;
static lv_obj_t *label_status;
static lv_obj_t *label_gain;
static lv_obj_t *btn_rec;
static lv_obj_t *btn_stop;

static bool recording_running = false;
static TaskHandle_t record_task_handle = NULL;
static volatile uint32_t recorded_bytes = 0;
static volatile int record_progress_percent = 0;

// ---------- WAV header util (16-bit PCM) ----------
void build_wav_header(uint8_t header[44], uint32_t sampleRate, uint16_t bitsPerSample, uint16_t channels, uint32_t dataBytes) {
  uint32_t byteRate = sampleRate * channels * bitsPerSample / 8;
  uint16_t blockAlign = channels * bitsPerSample / 8;
  uint32_t chunkSize = 36 + dataBytes;

  memset(header, 0, 44);
  header[0] = 'R'; header[1] = 'I'; header[2] = 'F'; header[3] = 'F';
  header[4] = (uint8_t)(chunkSize & 0xFF);
  header[5] = (uint8_t)((chunkSize >> 8) & 0xFF);
  header[6] = (uint8_t)((chunkSize >> 16) & 0xFF);
  header[7] = (uint8_t)((chunkSize >> 24) & 0xFF);
  header[8] = 'W'; header[9] = 'A'; header[10] = 'V'; header[11] = 'E';
  header[12] = 'f'; header[13] = 'm'; header[14] = 't'; header[15] = ' ';
  header[16] = 16; header[17] = 0; header[18] = 0; header[19] = 0;
  header[20] = 1; header[21] = 0;
  header[22] = (uint8_t)channels; header[23] = 0;
  header[24] = (uint8_t)(sampleRate & 0xFF);
  header[25] = (uint8_t)((sampleRate >> 8) & 0xFF);
  header[26] = (uint8_t)((sampleRate >> 16) & 0xFF);
  header[27] = (uint8_t)((sampleRate >> 24) & 0xFF);
  header[28] = (uint8_t)(byteRate & 0xFF);
  header[29] = (uint8_t)((byteRate >> 8) & 0xFF);
  header[30] = (uint8_t)((byteRate >> 16) & 0xFF);
  header[31] = (uint8_t)((byteRate >> 24) & 0xFF);
  header[32] = (uint8_t)(blockAlign & 0xFF);
  header[33] = (uint8_t)((blockAlign >> 8) & 0xFF);
  header[34] = (uint8_t)(bitsPerSample & 0xFF);
  header[35] = (uint8_t)((bitsPerSample >> 8) & 0xFF);
  header[36] = 'd'; header[37] = 'a'; header[38] = 't'; header[39] = 'a';
  header[40] = (uint8_t)(dataBytes & 0xFF);
  header[41] = (uint8_t)((dataBytes >> 8) & 0xFF);
  header[42] = (uint8_t)((dataBytes >> 16) & 0xFF);
  header[43] = (uint8_t)((dataBytes >> 24) & 0xFF);
}

// ----------------- I2S init (tu configuración funcional) -----------------
bool init_i2s_standard_rx() {
  i2s_driver_uninstall(I2S_NUM_0);

  i2s_config_t i2s_config = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = SAMPLE_RATE,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT,
    .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
    .communication_format = (i2s_comm_format_t)(I2S_COMM_FORMAT_STAND_I2S),
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,
    .dma_buf_len = 512,
    .use_apll = false,
    .tx_desc_auto_clear = false,
    .fixed_mclk = 0
  };

  esp_err_t err = i2s_driver_install(I2S_NUM_0, &i2s_config, 0, NULL);
  if (err != ESP_OK) {
    Serial.printf("i2s_driver_install err: %d\n", err);
    return false;
  }

  i2s_pin_config_t pin_config = {
    .bck_io_num = I2S_PIN_BCK,
    .ws_io_num  = I2S_PIN_WS,
    .data_out_num = I2S_PIN_DOUT,
    .data_in_num  = I2S_PIN_DIN
  };

  err = i2s_set_pin(I2S_NUM_0, &pin_config);
  if (err != ESP_OK) {
    Serial.printf("i2s_set_pin err: %d\n", err);
    return false;
  }

  err = i2s_set_clk(I2S_NUM_0, SAMPLE_RATE, I2S_BITS_PER_SAMPLE_32BIT, I2S_CHANNEL_STEREO);
  if (err != ESP_OK) {
    Serial.printf("i2s_set_clk err: %d\n", err);
    return false;
  }

  i2s_zero_dma_buffer(I2S_NUM_0);
  Serial.println("I2S configured OK");
  return true;
}

// ----------------- Recording task -----------------
void record_task(void *pvParameters) {
  // Abre archivo y escribe header temporal
  File f = SD_MMC.open(FILENAME, FILE_WRITE);
  if (!f) {
    Serial.println("ERROR: no pudo abrir archivo en SD");
    recording_running = false;
    vTaskDelete(NULL);
    return;
  }
  uint8_t zero44[44] = {0};
  f.write(zero44, 44);

  recorded_bytes = 0;
  uint8_t readBuf[I2S_READ_BYTES];
  uint32_t start = millis();
  while (recording_running && (millis() - start) < (uint32_t)RECORD_SECONDS * 1000) {
    size_t bytesRead = 0;
    esp_err_t r = i2s_read(I2S_NUM_0, readBuf, I2S_READ_BYTES, &bytesRead, portMAX_DELAY);
    if (r != ESP_OK) {
      Serial.printf("i2s_read err: %d\n", r);
      break;
    }
    if (bytesRead == 0) continue;

    int samples = bytesRead / 4;
    int32_t *words = (int32_t *)readBuf;

    for (int i = 0; i < samples; ++i) {
      int32_t w = words[i];
      int16_t sample16 = (int16_t)(w >> 16);
      // aplicar ganancia
      float amplified = (float)sample16 * g_mic_gain;
      if (amplified > 32767.0f) amplified = 32767.0f;
      if (amplified < -32768.0f) amplified = -32768.0f;
      int16_t out16 = (int16_t)amplified;
      f.write((uint8_t *)&out16, sizeof(out16));
      recorded_bytes += sizeof(out16);
    }
    // actualizar progreso aproximado
    uint32_t elapsed = millis() - start;
    record_progress_percent = (int)((elapsed * 100) / (RECORD_SECONDS * 1000));
  }

  // reconstruir header
  uint8_t header[44];
  build_wav_header(header, SAMPLE_RATE, 16 /*bitsPerSample*/, CHANNELS_OUT, recorded_bytes);
  f.seek(0);
  f.write(header, 44);
  f.close();

  recording_running = false;
  record_task_handle = NULL;
  record_progress_percent = 100;
  vTaskDelete(NULL);
}

// ----------------- LVGL: display flush and touch read -----------------
static void my_disp_flush(lv_disp_drv_t *drv, const lv_area_t *area, lv_color_t *color_p) {
  // area->x1..x2, area->y1..y2
  int x1 = area->x1;
  int y1 = area->y1;
  int x2 = area->x2;
  int y2 = area->y2;
  uint32_t w = (x2 - x1 + 1);
  uint32_t h = (y2 - y1 + 1);
  // lv_color_t is likely 16-bit. LCD_addWindow wants uint16_t*
  // Use the buffer directly (careful about endianness: your LCD_addWindow swaps bytes)
  uint16_t *buf16 = (uint16_t *)color_p;
  // Leverage your LCD_addWindow which expects (Xstart,Ystart,Xend,Yend,uint16_t* color)
  LCD_addWindow(x1, y1, x2, y2, buf16);
  lv_disp_flush_ready(drv);
}

static void touch_read_cb(lv_indev_drv_t * drv, lv_indev_data_t* data) {
  // Use Touch_Get_xy() from Touch_SPD2010 driver
  uint16_t tx = 0, ty = 0;
  uint8_t points = 0;
  bool pressed = Touch_Get_xy(&tx, &ty, NULL, &points, 1);
  if (pressed && points > 0) {
    // If touch coordinates differ in orientation, map/flip here.
    data->point.x = tx;
    data->point.y = ty;
    data->state = LV_INDEV_STATE_PR;
  } else {
    data->state = LV_INDEV_STATE_REL;
  }
}

// ----------------- UI events -----------------
static void event_rec_cb(lv_event_t * e) {
  if (recording_running) return;
  // start recording task
  recording_running = true;
  record_progress_percent = 0;
  xTaskCreatePinnedToCore(record_task, "recTask", 8 * 1024, NULL, 3, &record_task_handle, 1);
  lv_label_set_text(label_status, "Grabando...");
}

static void event_stop_cb(lv_event_t * e) {
  if (recording_running) {
    recording_running = false;
    // task will end; progress will be updated
    lv_label_set_text(label_status, "Deteniendo...");
  } else {
    lv_label_set_text(label_status, "No hay grabación");
  }
}

static void slider_event_cb(lv_event_t * e) {
  lv_obj_t *s = lv_event_get_target(e);
  int32_t v = lv_slider_get_value(s);
  // map slider to gain (ej: 1..8)
  float gain = 1.0f + (v / 10.0f); // simple mapping
  g_mic_gain = gain;
  char buf[32];
  snprintf(buf, sizeof(buf), "Ganancia: %.2fx", g_mic_gain);
  lv_label_set_text(label_gain, buf);
}

// ----------------- Create UI -----------------
void create_ui(void) {
  lv_obj_t *scr = lv_scr_act();

  // status label
  label_status = lv_label_create(scr);
  lv_obj_align(label_status, LV_ALIGN_TOP_MID, 0, 8);
  lv_label_set_text(label_status, "Listo. Presione GRABAR");

  // gain label
  label_gain = lv_label_create(scr);
  lv_obj_align(label_gain, LV_ALIGN_TOP_RIGHT, -8, 8);
  char buf[32];
  snprintf(buf, sizeof(buf), "Ganancia: %.2fx", g_mic_gain);
  lv_label_set_text(label_gain, buf);

  // Record button
  btn_rec = lv_btn_create(scr);
  lv_obj_set_size(btn_rec, 160, 80);
  lv_obj_align(btn_rec, LV_ALIGN_CENTER, -100, 40);
  lv_obj_add_event_cb(btn_rec, event_rec_cb, LV_EVENT_CLICKED, NULL);
  lv_obj_t *lab1 = lv_label_create(btn_rec);
  lv_label_set_text(lab1, "GRABAR");

  // Stop button
  btn_stop = lv_btn_create(scr);
  lv_obj_set_size(btn_stop, 160, 80);
  lv_obj_align(btn_stop, LV_ALIGN_CENTER, 100, 40);
  lv_obj_add_event_cb(btn_stop, event_stop_cb, LV_EVENT_CLICKED, NULL);
  lv_obj_t *lab2 = lv_label_create(btn_stop);
  lv_label_set_text(lab2, "DETENER");

  // Slider for gain (0..70 gives gain 1.0..8.0 roughly)
  lv_obj_t *s = lv_slider_create(scr);
  lv_obj_set_size(s, 300, 30);
  lv_obj_align(s, LV_ALIGN_BOTTOM_MID, 0, -30);
  lv_slider_set_range(s, 0, 70);
  lv_slider_set_value(s, (int)((g_mic_gain - 1.0f) * 10), LV_ANIM_OFF);
  lv_obj_add_event_cb(s, slider_event_cb, LV_EVENT_VALUE_CHANGED, NULL);
}

// ----------------- Setup / loop -----------------
void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println("\n=== INICIANDO SISTEMA I2S + LVGL + SPD2010 ===");

  // Inicializar Wire (I2C) para touch si lo usa
  Wire.begin();

  // Inicializar pantalla y touch (usa Display_SPD2010 driver que compartiste)
  // Tu Display_SPD2010.cpp define SPD2010_Init(), LCD_Init(), Backlight_Init(), Touch_Init()
  LCD_Init();
  if (!SPD2010_Init()) {
    Serial.println("Warning: SPD2010_Init() devolvió false (continuando)");
  }
  I2C_Init();
  TCA9554PWR_Init(0x00); 
  Backlight_Init();
  Set_Backlight(40);
  // Touch_Init();

  // Inicializar LVGL
  lv_init();

  // buffer para LVGL (allocate in DMA-capable memory, use PSRAM if available)
  size_t buf_pixels = LVGL_HOR_RES * 40; // 40 rows buffer (modificar si quieres más)
  lv_color_t *buf1 = (lv_color_t *)heap_caps_malloc(buf_pixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  lv_color_t *buf2 = (lv_color_t *)heap_caps_malloc(buf_pixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  static lv_disp_draw_buf_t disp_buf;
  lv_disp_draw_buf_init(&disp_buf, buf1, buf2, buf_pixels);

  // register display driver
  static lv_disp_drv_t disp_drv;
  lv_disp_drv_init(&disp_drv);
  disp_drv.hor_res = LVGL_HOR_RES;
  disp_drv.ver_res = LVGL_VER_RES;
  disp_drv.flush_cb = my_disp_flush;
  disp_drv.draw_buf = &disp_buf;
  lv_disp_t *d = lv_disp_drv_register(&disp_drv);
  (void)d;

  // register touch input device
  static lv_indev_drv_t indev_drv;
  lv_indev_drv_init(&indev_drv);
  indev_drv.type = LV_INDEV_TYPE_POINTER;
  indev_drv.read_cb = touch_read_cb;
  indev_touch = lv_indev_drv_register(&indev_drv);

  // Inicializar SD_MMC
  SD_MMC.setPins(SD_PIN_CLK, SD_PIN_CMD, SD_PIN_D0, -1, -1, -1);
  if (!SD_MMC.begin("/sdcard", true)) {
    Serial.println("ERROR: SD_MMC.begin() falló");
    while (true) { delay(1000); }
  }
  Serial.println("SD_MMC inicializada OK");

  // Inicializar I2S
  if (!init_i2s_standard_rx()) {
    Serial.println("ERROR: init I2S failed");
    while (true) { delay(1000); }
  }

  // Crear UI
  create_ui();

  Serial.println("Setup finalizado. UI y audio listos.");
}

// LVGL tick and handler: provide in loop
void loop() {
  // poll touch driver loop if needed (Touch_Loop does debug printing)
  // Touch_Loop(); // opcional, para debug

  // Update status label with progress
  static uint32_t last_update = 0;
  if (millis() - last_update > 200) {
    if (recording_running) {
      char s[64];
      snprintf(s, sizeof(s), "Grabando... %d%%", record_progress_percent);
      lv_label_set_text(label_status, s);
    } else {
      if (record_progress_percent == 100) {
        lv_label_set_text(label_status, "Grabacion OK");
        record_progress_percent = 0;
      }
    }
    last_update = millis();
  }

  lv_timer_handler(); // required for LVGL
  delay(5);
}
